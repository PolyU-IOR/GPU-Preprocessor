cmake_minimum_required(VERSION 3.24)
project(GPUPreprocessor VERSION 2.0.0 LANGUAGES C CXX)
include(GNUInstallDirs)
# Preserve LP option names and accept the native QP build switches.
set(tools_default ${PROJECT_IS_TOP_LEVEL})
set(examples_default ${PROJECT_IS_TOP_LEVEL})
set(tests_default OFF)
if(DEFINED BUILD_TOOLS)
  set(tools_default ${BUILD_TOOLS})
endif()
if(DEFINED BUILD_EXAMPLES)
  set(examples_default ${BUILD_EXAMPLES})
endif()
if(DEFINED BUILD_TESTING)
  set(tests_default ${BUILD_TESTING})
endif()
option(GPUPREPROCESSOR_BUILD_TOOLS "Build command-line programs" ${tools_default})
option(GPUPREPROCESSOR_BUILD_EXAMPLES "Build examples" ${examples_default})
option(GPUPREPROCESSOR_BUILD_TESTS "Build regression tests" ${tests_default})
set(BUILD_TOOLS ${GPUPREPROCESSOR_BUILD_TOOLS})
set(BUILD_EXAMPLES ${GPUPREPROCESSOR_BUILD_EXAMPLES})
set(BUILD_TESTING ${GPUPREPROCESSOR_BUILD_TESTS})
include(CTest)
if(DEFINED ENABLE_CUDA AND NOT ENABLE_CUDA)
  message(FATAL_ERROR "CPU-only builds were removed. LP and QP require CUDA; use a fresh build directory.")
endif()
set(ENABLE_CUDA ON)
option(GPUPREPROCESSOR_BUILD_LP "Build the independent GPU LP module" ON)
option(GPUPREPROCESSOR_BUILD_QP "Build the QP module and its applications" ON)
option(GPUPREPROCESSOR_USE_INT64_NNZ "Use int64 LP sparse offsets; dimensions stay int32" OFF)
option(BUILD_SHARED_LIBS "Build shared native QP libraries" ON)
if(NOT GPUPREPROCESSOR_BUILD_LP AND NOT GPUPREPROCESSOR_BUILD_QP)
  message(FATAL_ERROR "Enable at least one of LP or QP")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib")
set(GPUPREPROCESSOR_NATIVE_CORE OFF)
if(GPUPREPROCESSOR_BUILD_QP)
  set(GPUPREPROCESSOR_NATIVE_CORE ON)
  find_package(ZLIB REQUIRED)
endif()
find_package(Threads REQUIRED)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND "$ENV{CUDAARCHS}" STREQUAL "")
  set(CMAKE_CUDA_ARCHITECTURES native CACHE STRING "CUDA architectures")
endif()
enable_language(CUDA)
find_package(CUDAToolkit 12 REQUIRED)
function(warnings target)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic>)
endfunction()
function(application_runtime target)
  if(APPLE)
    set_target_properties(${target} PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
  elseif(UNIX)
    set_target_properties(${target} PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  endif()
endfunction()
if(GPUPREPROCESSOR_BUILD_LP)
  add_subdirectory(lp)
endif()
if(GPUPREPROCESSOR_NATIVE_CORE)
  add_subdirectory(qp)
endif()
if(BUILD_TESTING AND GPUPREPROCESSOR_BUILD_LP AND GPUPREPROCESSOR_BUILD_QP)
  add_executable(component_link_tests tests/component_link.cpp)
  target_link_libraries(component_link_tests PRIVATE GPUPreprocessor::lp GPUPreprocessor::qp)
  add_test(NAME component_link_cpu COMMAND component_link_tests)
  set_tests_properties(component_link_cpu PROPERTIES LABELS "cpu;integration")
endif()
set(STATIC_CUDA OFF)
if(NOT BUILD_SHARED_LIBS AND GPUPREPROCESSOR_NATIVE_CORE)
  set(STATIC_CUDA ON)
endif()
include(cmake/Install.cmake)
